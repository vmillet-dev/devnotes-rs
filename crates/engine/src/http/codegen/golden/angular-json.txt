import { HttpHeaders } from '@angular/common/http';

this.http
  .request('POST', 'https://api.exemple.fr/customers', {
    headers: new HttpHeaders({
      'Content-Type': 'application/json',
    }),
    body: `{
  "name": "O'Brien",
  "plan": "pro"
}`,
    responseType: 'text',
  })
  .subscribe((response) => console.log(response));
