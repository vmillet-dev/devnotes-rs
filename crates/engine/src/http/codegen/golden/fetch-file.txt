import { openAsBlob } from 'node:fs';

const response = await fetch('https://api.exemple.fr/upload', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/octet-stream',
  },
  body: await openAsBlob('/tmp/archive.zip'),
  redirect: 'manual',
});
console.log(response.status, await response.text());
