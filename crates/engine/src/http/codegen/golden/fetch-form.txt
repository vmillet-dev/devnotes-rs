const response = await fetch('https://auth.exemple.fr/token', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/x-www-form-urlencoded',
    Authorization: `Basic ${btoa('billing-web:{{clientSecret}}')}`,
  },
  body: new URLSearchParams({
    grant_type: 'client_credentials',
    scope: 'invoices read',
  }),
  redirect: 'manual',
});
console.log(response.status, await response.text());
