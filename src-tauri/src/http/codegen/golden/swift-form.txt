import Foundation

var request = URLRequest(url: URL(string: "https://auth.exemple.fr/token")!)
request.httpMethod = "POST"
request.setValue("application/x-www-form-urlencoded", forHTTPHeaderField: "Content-Type")
request.setValue("Basic " + Data("billing-web:{{clientSecret}}".utf8).base64EncodedString(), forHTTPHeaderField: "Authorization")
request.httpBody = Data("grant_type=client_credentials&scope=invoices+read".utf8)
let (data, response) = try await URLSession.shared.data(for: request)
print((response as! HTTPURLResponse).statusCode, String(decoding: data, as: UTF8.self))
